const Cube ={
    uniforms: {
        iResolution: Common.u_Resolution, 
        iMouse: Common.u_Mouse, 
        u_Pos: {value: 0.0}, 
        u_Rot: Common.u_Rotation
    }, 
    vertexShader: `
    void main (){
        gl_Position =vec4(position, 1.0);
    }`, 
    fragmentShader: `
    precision mediump float;
    
    #define MAX_STEPS 100
    #define MAX_DIST 100.0
    #define SURF_DIST 0.001
    #define PI 3.14159265358979
    #define S (a, b, t) smoothstep(a, b, t)

    uniform vec2 iResolution;
    uniform vec2 iMouse;
    uniform vec3 u_Pos;
    uniform vec2 u_Rot;

    mat2 Rot (float a){
        float s =sin(a);
        float c =cos(a);
        
        return mat2 (c, -s, s, c);
    }

    float SdBox (vec3 p, vec3 s){
        p =abs(p) -s;
        return length(max(p, 0.0)) +min(max(p.x, max(p.y, p.z)), 0.0);
    }

    float GetDist (vec3 p){
        float box =SdBox (p -vec3(0.0, 1.0, 0.0), vec3(1.0));
        //float giroid =dot(sin(p), cos(p.zxy));

        return box;
    }

    float RayMarch (vec3 ro, vec3 rd){
        float d0 =0.0;

        for (int i =0;  i < MAX_STEPS; i++){
            vec3 p =ro +rd*d0;
            float dS =GetDist (p);
            d0 +=dS;
            if (d0 > MAX_DIST || abs(dS) < SURF_DIST) break;
        }

        return d0;
    }

    vec3 GetNormal (vec3 p){
        float d =GetDist (p);
        vec2 e =vec2(0.001, 0);

        vec3 n =d -vec3(GetDist (p -e.xyy), GetDist (p -e.yxy), GetDist (p -e.yyx));

        return normalize(n);
    }

    vec3 GetRayDir (vec2 uv, vec3 p, vec3 l, float z){
        vec3 f =normalize(l -p),
        r =normalize(cross(vec3(0, 1, 0), f)), 
        u =cross(f, r), 
        c =p +f*z, 
        i =c +uv.x*r +uv.y*u, 
        d =normalize(i -p);

        return d;
    }

    void main (){
        vec2 uv =(gl_FragCoord.xy -iResolution.xy*0.5)/iResolution.y;
        vec2 m =2.5*u_Rot.yx/iResolution.xy;
        vec3 col =vec3(0.0);

        vec3 ro =vec3(0, 3, -3);
        ro.yz *=Rot (-m.y*1.0*PI +1.0);
        ro.xz *=Rot (-m.x*2.0*PI);

        vec3 rd =GetRayDir (uv, ro, vec3(0, 1, 0), 1.0);
        float d =RayMarch (ro, rd);

        if (d < MAX_DIST){
            vec3 p =ro +rd*d;
            vec3 n =GetNormal (p);
            float dif =dot(n, normalize(vec3(1, 2, 3)))*0.5 +0.5;
            col +=dif;
        }

        gl_FragColor =vec4(col, 1.0);
    }`
}
